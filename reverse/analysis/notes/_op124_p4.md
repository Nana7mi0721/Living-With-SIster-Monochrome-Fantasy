
### 6.3 `char_get`（0x1000 / 0x1200）

```
fn char_get(vm: &Vm, obj: i64, field: i64, zoom: bool) -> Option<i64> {
    // 1) 解析出「角色」；解析失败（事件不存在）=> None => 不写变量
    let ch = match obj {
        id if id >= 0     => vm.map.event_by_id(id as u32)?,   // 地图事件
        -1                => vm.interp.current_char_ref()?,     // 実行中イベント；コモンで呼出元無し => 全値 -1
        -2                => vm.party.leader.clone(),
        -3..=-7           => vm.party.member((-obj - 3) as usize)?,
        _                 => return None,
    };
    Some(match field {
        0 => ch.tile_x() as i64,                     // 標準（マス）
        1 => ch.tile_y() as i64,
        2 => ch.precise_x() as i64,                  // 精密 = 0.5マス = 標準*2
        3 => ch.precise_y() as i64,
        4 => ch.height_px() as i64,
        5 => ch.direction() as i64,                  // 1..9
        6 => if zoom { ch.screen_x_zoomed(vm) } else { ch.screen_x() },          // 中央
        7 => if zoom { ch.screen_y_zoomed(vm) } else { ch.screen_y() },          // 足下
        8 => ch.screen_x_zoomed(vm),
        9 => ch.screen_y_zoomed(vm),
        10 => ch.event_id() as i64,                  // 本作ここ = イベントID（新版は 12）
        _ => { tracing::debug!("124/0x1000 field {field} unused"); 0 }
    })
}
```

`screen_x()` = キャラ左右中央の画面X、`screen_y()` = **足下**の画面Y（官方注記）。
ズーム反映版：`zoom_origin + (screen - zoom_origin) * zoom_pct / 100`；
本作タイトル画面の zoom = 100% ⇒ 恒等。

**一个必须做对的细节**：`obj == -1`（実行中イベント）且当前是**没有调用元**的公共事件时，
官方行为是「全属性都写入 -1」，**不是**「不写变量」。所以 `current_char_ref()` 返回 None 时
应当 `Some(-1)`，而「地图事件不存在」才是 `None`。

### 6.4 `pic_get`（0x4000 / 0x4400）

```
fn pic_get(vm: &Vm, pic: i64, field: i64, no_effect: bool) -> Option<i64> {
    let p = vm.pictures.get(pic);          // Option<&Picture>，未使用的番号 => None
    Some(match field {
        0 => p.map_or(-1, |p| p.draw_x(no_effect)),      // X座標（補正あり/なし）
        1 => p.map_or(-1, |p| p.draw_y(no_effect)),      // Y座標
        2 => p.map_or(-1, |p| p.scale_x()),              // 拡大率(横)   … §9.2
        3 => p.map_or(-1, |p| p.scale_y()),              // 拡大率(縦)
        4 => p.map_or(-1, |p| p.pattern()),              // 現在のパターン番号
        5 => 0,                                          // 未定（§9.3）
        8 => {                                           // マウスカーソル重なってる？
            let Some(p) = p else { return Some(-1) };    // 画像が無い => -1
            hit_test(vm, p) as i64                       // 1 / 0
        }
        9 => p.is_some() as i64,                         // ピクチャが使用されている
        10 => match p {                                  // 文字列、表示完了？
            None => -1,
            Some(p) if !p.is_text() => -1,
            Some(p) => if p.text_finished() { 1 } else { 0 },
        },
        _ => { tracing::debug!("124/0x4000 field {field} unused"); 0 }
    })
}
```

### 6.5 悬停判定 `hit_test`（field 8，588 处调用）

```
fn hit_test(vm: &Vm, p: &Picture) -> bool {
    let (mx, my) = (vm.input.mouse_x(), vm.input.mouse_y());
    if !(0..1024).contains(&mx) || !(0..576).contains(&my) { return false; }  // 屏幕外

    let w = p.src_w * p.scale_x / 100;
    let h = p.src_h * p.scale_y / 100;
    let (x, y) = (p.draw_x(false), p.draw_y(false));

    if p.angle == 0 && !p.free_transform {
        // 简化版：AABB（本作按钮几乎都是未旋转矩形）
        mx >= x && mx < x + w && my >= y && my < y + h
    } else {
        // 旋转/自由変形：四角点内判定（官方：判定範囲は多少広くなる）
        point_in_quad((mx, my), p.corners())
    }
    // 注：透明ピクセルも「重なっている」と判定する（官方注記）。
    //     像素级 alpha 测试不做。
}
```
